// The response rules' root file, loaded by the addon's autorun
// (lua/autorun/server/responserules.lua) from scripts/responserules/ on the
// GAME path, in the grammar of TF2's scripts/talker/response_rules.txt,
// which documents it. The criteria the rules share are here, as in TF2's;
// the rules themselves are by the speaker model's affiliation (HL2's sides),
// one file each, included below (paths relative to scripts/).
//
// Not played here, and refused when loaded: scene and sentence lines,
// predelay, weapondelay, stop_on_nonidle, displayfirst, displaylast,
// applyContextToWorld.
//
// A speaker's criteria set, as Sandcastle's AI war gives it
// (features/aiwar/sv_speech.lua, foundation.aiwar.speech):
//   affiliation  combine, rebel (foundation.npc.voices)
//   voice        combine, metrocop, male, female, alyx, vortigaunt
//   teamrole     offense (the cart's own team), defense
//   randomnum    0..100
//   cartblocked  1 while the enemy holds the cart
// plus the contexts its rules applied.

// a teammate player passing the soldier, and an enemy it has just spotted
criterion "ConceptPlayerPassing" "Concept" "TLK_NS_PLAYER_PASSING" required
criterion "ConceptEnemySpotted" "Concept" "TLK_NS_ENEMY_SPOTTED" required
// a hurt soldier calling for a medic
criterion "ConceptCallMedic" "Concept" "TLK_NS_CALL_MEDIC" required
// a hurt soldier falling back to cover, and leaving for a health kit
criterion "ConceptRetreatToCover" "Concept" "TLK_NS_RETREAT_TO_COVER" required
criterion "ConceptGetHealth" "Concept" "TLK_NS_GET_HEALTH" required
// squad chatter: a kill, a teammate down near it, an enemy slipped out of
// sight, and the radio between fights
criterion "ConceptEnemyKilled" "Concept" "TLK_NS_ENEMY_KILLED" required
criterion "ConceptAllyDown" "Concept" "TLK_NS_ALLY_DOWN" required
criterion "ConceptContactLost" "Concept" "TLK_NS_CONTACT_LOST" required

// a squad leader calling its squad out on its mission

// a squad leader calling its squad together to a hiding spot, its mission gone wrong
criterion "ConceptRegroup" "Concept" "TLK_NS_REGROUP" required
// a soldier taunting, a long line to go with the taunt
criterion "ConceptTaunt" "Concept" "TLK_NS_TAUNT" required

criterion "CartBlocked" "cartblocked" "1" required weight 2
