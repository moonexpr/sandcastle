// Combine: the cart rules for the Combine's models: its soldiers, its metrocops.
// A model answers by its affiliation first, whichever team it fights for:
// each situation has a rule for the affiliation (placeholder lines, shown
// with responserules_debug 1), and one for each voice with lines of its own,
// which outscores it by its voice criterion. A new model of the Combine
// speaks by these rules as soon as foundation.npc.voices gives it its affiliation.
// The criteria the cart's situations share are in response_rules.txt.

criterion "IsCombine" "affiliation" "combine" required
criterion "IsSoldierVoice" "voice" "combine" required
criterion "IsMetrocopVoice" "voice" "metrocop" required

// CartMovingForwardOffense: the cart's moving: keep pushing
response CartMovingForwardOffenseCombine
{
	print "Combine: the cart's moving: keep pushing" defaultdelay
}
rule CartMovingForwardOffenseCombine
{
	criteria ConceptCartMovingForward IsOnOffense IsCombine NotSaidCartMovingForwardO 75PercentChance
	ApplyContext "SaidCartMovingForwardO:1:20"
	response CartMovingForwardOffenseCombine
}
response CartMovingForwardOffenseCombineSoldier
{
	speak "npc/combine_soldier/vo/movein.wav" defaultdelay
	speak "npc/combine_soldier/vo/sweepingin.wav" defaultdelay
	speak "npc/combine_soldier/vo/closing.wav" defaultdelay
}
rule CartMovingForwardOffenseCombineSoldier
{
	criteria ConceptCartMovingForward IsOnOffense IsCombine NotSaidCartMovingForwardO 75PercentChance IsSoldierVoice
	ApplyContext "SaidCartMovingForwardO:1:20"
	response CartMovingForwardOffenseCombineSoldier
}
response CartMovingForwardOffenseCombineMetrocop
{
	speak "npc/metropolice/vo/moveit.wav" defaultdelay
	speak "npc/metropolice/vo/allunitsmovein.wav" defaultdelay
}
rule CartMovingForwardOffenseCombineMetrocop
{
	criteria ConceptCartMovingForward IsOnOffense IsCombine NotSaidCartMovingForwardO 75PercentChance IsMetrocopVoice
	ApplyContext "SaidCartMovingForwardO:1:20"
	response CartMovingForwardOffenseCombineMetrocop
}
